@startuml SkyTrip_Sequence_Login

title Sequence Diagram — User Login & Profile Load

actor User
participant "AuthScreen" as Auth
participant "AuthService" as AS
participant "Azure API" as API
participant "UserProvider" as UP
participant "ProfileService" as PS
participant "MainScaffold" as Main

User -> Auth : Enter email & password
Auth -> Auth : Validate input fields

alt Validation fails
  Auth --> User : Show validation error
else Validation passes
  Auth -> AS : login(email, password)
  AS -> API : POST /api/Login/login\n{email, password}

  alt 401 / Network Error
    API --> AS : Error response
    AS --> Auth : throw AuthException
    Auth --> User : Show error SnackBar
  else 200 OK
    API --> AS : {token, userId, personId,\nclientId, email, role}
    AS --> Auth : LoginResult
    Auth -> UP : login(token, userId, personId,\nclientId, email, role)
    UP -> UP : Store auth state\n+ notifyListeners()

    Auth -> UP : loadProfile(token)
    UP -> PS : getMyProfile(token)
    PS -> API : GET /api/profile/me
    API --> PS : ProfileData
    PS --> UP : ProfileData
    UP -> UP : Update firstName, lastName,\nphone, nationality, etc.

    Auth -> Main : Navigator.pushReplacementNamed('/home')
    Main --> User : Show Home Screen
  end
end

  PFS -> BKS : addPassenger(bookId, formData, docFile, token)

@startuml SkyTrip_Sequence_Payment

title Sequence Diagram — Payment & Confirmation

actor User
participant "PaymentScreen" as PS
participant "PaymentService" as PAS
participant "Stripe" as Stripe
participant "PaymentSuccessScreen" as PSS
participant "BookingProvider" as BP
participant "Azure API" as API
participant "TicketsScreen" as TS

== Step 7: Initiate Payment ==

PS --> User : Show price breakdown\n(baseFare + taxes + services)
User -> PS : Tap "Pay Now"

PS -> PAS : createCheckoutSession(ticketId, token)
PAS -> API : POST /api/payments/checkout-session\n{ticketId}
API --> PAS : Stripe checkout URL
PAS --> PS : checkoutUrl

PS -> Stripe : launchUrl(checkoutUrl)\n(Chrome Custom Tab)
Stripe --> User : Stripe payment page

alt User completes payment
  User -> Stripe : Enter card details & pay
  Stripe -> Stripe : Process payment
  Stripe --> User : Redirect via deep link\nskytrip://payment-success?ticketId=XXX

  == Step 8: Confirm Payment ==

  User -> PSS : Deep link opens app
  PSS -> PSS : Start polling timer (every 1.5s)

  loop Poll up to 25 seconds
    PSS -> PAS : getPaymentStatus(ticketId, token)
    PAS -> API : GET /api/payments/my/status/{ticketId}
    API --> PAS : {isConfirmed, paymentStatus,\nbookingStatus}

    alt Status == "Paid"
      PAS --> PSS : "Paid"
      PSS --> User : Show success animation\n& booking reference
      PSS -> BP : resetBooking()
      User -> TS : Navigate to /tickets
    else Status == "Pending"
      PAS --> PSS : "Pending"
      PSS -> PSS : Continue polling
    end
  end

else User cancels
  Stripe --> User : Redirect via deep link\nskytrip://payment-cancel
  User -> PS : Show cancellation screen\nwith "Retry" button
end

@enduml

title Sequence Diagram — Employee Dashboard Operations

actor "Airline Employee" as Emp
participant "Employee.html\n(Web Dashboard)" as EMP
participant "Azure API" as API

Emp -> EMP : Login (token from Login.html)
EMP -> EMP : Read token from localStorage
EMP -> API : GET /api/UserAccount/me
API --> EMP : User info (name, role)

EMP -> API : GET /api/FlightSchedules/All
EMP -> API : GET /api/BookingTrip
EMP -> API : GET /api/Services/All
EMP -> API : GET /api/FlightSchedules/Cities
API --> EMP : Flights, Bookings,\nServices, Cities data
EMP --> Emp : Render dashboard\n(stats + flights table)

== Update Flight Schedule ==

Emp -> EMP : Click "Update" on a flight row
EMP --> Emp : Open flight editor drawer\n(pre-filled with current data)
Emp -> EMP : Modify fields\n(cities, date, times, base price)
Emp -> EMP : Click "Save changes"
EMP -> API : PUT /api/FlightSchedules/{id}\n{departureCity, arrivalCity,\ndepartureTime, arrivalTime,\nflightDate, basePrice}
API --> EMP : 200 OK
EMP --> Emp : Toast "Flight updated successfully"
EMP -> EMP : Reload dashboard data

== Review Booking (Passengers & Documents) ==

Emp -> EMP : Switch to Bookings tab
EMP --> Emp : Display bookings table\n(filterable: All / Pending / Confirmed)
Emp -> EMP : Click "Review" on a booking
EMP -> API : GET /api/bookings/{bookId}/passengers
API --> EMP : {bookId, ticketId,\nexpectedPassengers, createdPassengers,\npassengers: [{name, gender, birthDate,\ndocumentationType, issueCountryId,\nexpirationDate, hasDocument,\ndocumentImageBase64}]}
EMP --> Emp : Open review modal\nwith passenger cards\n+ document previews\n(image / PDF)

== Confirm Booking ==

Emp -> EMP : Click "Confirm" on a booking\n(from table or review modal)
EMP -> API : POST /api/BookingTrip/{bookId}/confirm
API --> EMP : 200 OK
EMP --> Emp : Toast "Booking confirmed"
EMP -> EMP : Reload dashboard data

== View Ticket Info ==

Emp -> EMP : Click "View" on a booking
EMP -> API : GET /api/InfoTickets/ByBooking/{bookId}
API --> EMP : List of ticket info\n(ticketId, passengerClassId,\nticketPrice, passengersCount)
EMP --> Emp : Open drawer with\nticket info cards

Adm -> ADM : Confirm deletion
ADM -> API : DELETE /api/FlightSchedules/{id}
